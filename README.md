    const categoryNames = {
      pen: "خودکار و مداد",

    // State
    let products = [];
    let isAdmin = false;
    let currentFilter = "all";
    let editingId = null; // null = add new

    // Load products from localStorage or defaults
    function loadProducts() {
      const saved = localStorage.getItem("ali_stationery_products");
      if (saved) {
        try {
          products = JSON.parse(saved);
        } catch {
          products = [...defaultProducts];
        }
      } else {
        products = [...defaultProducts];
      }
    }

    function saveProducts() {
      localStorage.setItem("ali_stationery_products", JSON.stringify(products));
    }

    function formatPrice(price) {
      return Number(price).toLocaleString("fa-IR") + " تومان";
    }

    function showToast(msg) {
      const t = document.getElementById("toast");
      t.textContent = msg;
      t.classList.add("show");
      setTimeout(() => t.classList.remove("show"), 2800);
    }

    // Render products
    function renderProducts(filter = currentFilter) {
      currentFilter = filter;
      const container = document.getElementById("products");
      const filtered = filter === "all"
        ? products
        : products.filter(p => p.category === filter);

      let html = filtered.map(p => `
        <div class="product-card" data-id="${p.id}">
          <div class="product-image">
            ${p.image
              ? `<img src="${p.image}" alt="${p.name}" onerror="this.style.display='none';this.nextElementSibling.style.display='flex'">
                 <div class="placeholder" style="display:none">${p.emoji || "📦"}</div>`
              : `<div class="placeholder">${p.emoji || "📦"}</div>`
            }
            ${p.badge ? `<span class="product-badge">${p.badge}</span>` : ""}
            <button class="admin-edit-btn" data-edit="${p.id}" title="ویرایش">✏️</button>
          </div>
          <div class="product-body">
            <div class="product-category">${p.categoryName || categoryNames[p.category]}</div>
            <div class="product-name">${p.name}</div>
            <div class="product-price">
              <div class="price">${formatPrice(p.price)}</div>
              <button class="view-btn">مشاهده</button>
            </div>
          </div>
        </div>
      `).join("");

      // Add product card (only visible in admin mode via CSS)
      html += `
        <div class="product-card add-product-card" id="addProductCard">
          <span>➕</span>
          <p>افزودن محصول جدید</p>
        </div>
      `;

      container.innerHTML = html;

      // Click on product card → open detail
      container.querySelectorAll(".product-card:not(.add-product-card)").forEach(card => {
        card.addEventListener("click", (e) => {
          if (e.target.closest(".admin-edit-btn")) return;
          openModal(parseInt(card.dataset.id));
        });
      });

      // Edit buttons
      container.querySelectorAll(".admin-edit-btn").forEach(btn => {
        btn.addEventListener("click", (e) => {
          e.stopPropagation();
          openEditModal(parseInt(btn.dataset.edit));

      const addCard = document.getElementById("addProductCard");
      if (addCard) {
        addCard.addEventListener("click", () => openEditModal(null));

    function openEditModal(id) {

    document.getElementById("saveProduct").addEventListener("click", () => {
      const name = document.getElementById("editName").value.trim();
      const category = document.getElementById("editCategory").value;
      const price = parseInt(document.getElementById("editPrice").value) || 0;
      const emoji = document.getElementById("editEmoji").value.trim() || "📦";
      const badge = document.getElementById("editBadge").value.trim();
      const image = document.getElementById("editImage").value.trim();
      const description = document.getElementById("editDesc").value.trim();

      if (!name) {
        alert("نام محصول را وارد کنید");
        return;
      }
      if (price <= 0) {
        alert("قیمت معتبر وارد کنید");
        return;
      }

      if (editingId === null) {
        // Add new
        const newId = products.length ? Math.max(...products.map(p => p.id)) + 1 : 1;
        products.push({
          id: newId,
          name,
          category,
          categoryName: categoryNames[category],
          price,
          emoji,
          image,
          badge,
          description
        });
        showToast("محصول جدید اضافه شد ✓");
      } else {
        // Update
        const idx = products.findIndex(p => p.id === editingId);
        if (idx !== -1) {
          products[idx] = {
            ...products[idx],
            name,
            category,
            categoryName: categoryNames[category],
            price,
            emoji,
            image,
            badge,
            description
          };
          showToast("محصول به‌روزرسانی شد ✓");
        }
      }

      saveProducts();
      renderProducts();
      closeEditModal();
    });

    document.getElementById("deleteProduct").addEventListener("click", () => {
      if (editingId === null) return;
      if (!confirm("آیا از حذف این محصول مطمئن هستید؟")) return;

      products = products.filter(p => p.id !== editingId);
      saveProducts();
      renderProducts();
      closeEditModal();
      showToast("محصول حذف شد");
    });

    // Init
    loadProducts();
    renderProducts();
